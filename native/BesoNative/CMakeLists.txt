cmake_minimum_required(VERSION 3.16)
project(BesoNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
find_package(Eigen3 REQUIRED NO_MODULE)
find_path(CHOLMOD_INCLUDE_DIR cholmod.h HINTS /opt/homebrew/opt/suite-sparse/include/suitesparse /usr/local/include/suitesparse /usr/include/suitesparse REQUIRED)
add_library(BesoNative SHARED solver.cpp)
target_include_directories(BesoNative PRIVATE ${CHOLMOD_INCLUDE_DIR})
target_link_libraries(BesoNative PRIVATE Eigen3::Eigen)
foreach(name cholmod amd camd colamd ccolamd suitesparseconfig)
  find_library(SS_${name} NAMES ${name} HINTS /opt/homebrew/opt/suite-sparse/lib REQUIRED)
  target_link_libraries(BesoNative PRIVATE ${SS_${name}})
endforeach()
if(APPLE)
  target_link_libraries(BesoNative PRIVATE "-framework Accelerate")
else()
  find_package(BLAS REQUIRED)
  find_package(LAPACK REQUIRED)
  target_link_libraries(BesoNative PRIVATE ${BLAS_LIBRARIES} ${LAPACK_LIBRARIES})
endif()
set_target_properties(BesoNative PROPERTIES PREFIX "" SUFFIX ".dll")
